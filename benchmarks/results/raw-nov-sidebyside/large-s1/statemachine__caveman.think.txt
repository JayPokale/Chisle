class OrderState(Enum):

    CANCELLED = "cancelled"
    REFUNDED = "refunded"

class OrderStateMachine:
    # Define valid transitions as {from_state: {to_state}}
    VALID_TRANSITIONS: Dict[OrderState, Set[OrderState]] = {
        OrderState.CREATED: {OrderState.PAID, OrderState.CANCELLED},
        OrderState.PAID: {OrderState.SHIPPED, OrderState.REFUNDED, OrderState.CANCELLED},
        OrderState.SHIPPED: {OrderState.DELIVERED, OrderState.REFUNDED, OrderState.CANCELLED},
        OrderState.DELIVERED: {OrderState.REFUNDED},
        OrderState.CANCELLED: set(),
        OrderState.REFUNDED: set(),

    def __init__(self, initial_state: OrderState = OrderState.CREATED):

    def can_transition(self, next_state: OrderState) -> bool:
        return next_state in self.VALID_TRANSITIONS[self.current_state]
