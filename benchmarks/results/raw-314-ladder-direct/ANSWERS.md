> Parse --port and --verbose command-line flags in a Node.js script. Show the code.

### vanilla: 403 answer tokens, 726 billed, 57 lines

Here's a simple way to parse `--port` and `--verbose` flags in Node.js:

```javascript
const args = process.argv.slice(2);
let port = 3000;
let verbose = false;

for (let i = 0; i < args.length; i++) {
  if (args[i] === '--port' && args[i + 1]) {
    port = parseInt(args[i + 1], 10);
    i++;
  } else if (args[i] === '--verbose') {
    verbose = true;
  }
}

console.log('Port:', port);
console.log('Verbose:', verbose);
```

Usage:
```bash
node script.js --port 8080 --verbose
# Port: 8080
# Verbose: true
```

**Alternative: using a library**

If you prefer cleaner syntax, use `minimist`:

```javascript
const minimist = require('minimist');
const argv = minimist(process.argv.slice(2), {
  default: { port: 3000, verbose: false }
});

console.log('Port:', argv.port);
console.log('Verbose:', argv.verbose);
```

Or with `commander`:

```javascript

program
  .option('--port <number>', 'port number', '3000')
  .option('--verbose', 'verbose output')
  .parse(process.argv);

const options = program.opts();
console.log('Port:', options.port);
console.log('Verbose:', options.verbose);
```

The manual approach works fine for simple cases; use a library for more complex argument parsing.

Here's how to convert an ISO timestamp to a localized date format in your web app:

**Using `Intl.DateTimeFormat` (no dependencies):**
```javascript
function formatDate(isoTimestamp) {
  const date = new Date(isoTimestamp);
  return new Intl.DateTimeFormat('en-US', {
    year: 'numeric',
    month: 'short',
    day: 'numeric'
  }).format(date);
}

// Usage
formatDate('2026-11-23T14:30:00Z'); // "Nov 23, 2026"
```

**Using `toLocaleDateString` (simpler but less control):**
```javascript
function formatDate(isoTimestamp) {
  return new Date(isoTimestamp).toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric'
  });
}
```

**In React:**
```jsx
function DateDisplay({ timestamp }) {
  const formatted = new Date(timestamp).toLocaleDateString('en-US', {
    month: 'short',

}
```

**With date-fns (if you're already using it):**

> Remove duplicate emails from a list of users, keeping the first occurrence and the original order. Show the code.

### vanilla: 484 answer tokens, 680 billed, 52 lines

Here's a clean solution in Python that removes duplicate emails while preserving order:

```python
def remove_duplicate_emails(users):
    """Remove duplicate emails, keeping first occurrence and original order."""
    seen = set()
    result = []
    for user in users:
        if user['email'] not in seen:
            seen.add(user['email'])
            result.append(user)
    return result

# Example usage
users = [
    {'name': 'Alice', 'email': 'alice@example.com'},
    {'name': 'Bob', 'email': 'bob@example.com'},
    {'name': 'Alice2', 'email': 'alice@example.com'},  # duplicate
    {'name': 'Charlie', 'email': 'charlie@example.com'},
    {'name': 'Bob2', 'email': 'bob@example.com'},  # duplicate
]

result = remove_duplicate_emails(users)
for user in result:
    print(user)
```

Output:
```
{'name': 'Alice', 'email': 'alice@example.com'}
{'name': 'Bob', 'email': 'bob@example.com'}
{'name': 'Charlie', 'email': 'charlie@example.com'}
```

**Alternative (more concise):**
```python
def remove_duplicate_emails(users):
    seen = set()
    return [u for u in users if not (u['email'] in seen or seen.add(u['email']))]
```

**If you're working with just email strings:**
```python
def remove_duplicate_emails(emails):
    seen = set()
    return [e for e in emails if not (e in seen or seen.add(e))]
```

The approach:
- Use a `set()` to track seen emails in O(1) lookup time
- Iterate through the list once, keeping only emails we haven't seen yet
- Time complexity: O(n), Space complexity: O(n)

# Deep Copy of Nested Settings Object

Here are implementations in common languages:

## JavaScript

```javascript
// Original settings object
const originalSettings = {
  theme: { mode: 'dark', colors: { primary: '#000' } },
  notifications: { email: true, sms: false }
};

// Deep copy using structuredClone (recommended, modern)
const settingsCopy = structuredClone(originalSettings);

// Or using JSON (works for plain objects, not functions/symbols)
const settingsCopy = JSON.parse(JSON.stringify(originalSettings));

// Now edit the copy safely
settingsCopy.theme.colors.primary = '#fff';
settingsCopy.notifications.email = false;

console.log(originalSettings.theme.colors.primary); // '#000' (unchanged)
console.log(settingsCopy.theme.colors.primary);     // '#fff' (changed)
```

## Python

```python
import copy

# Original settings object
original_settings = {
    'theme': {'mode': 'dark', 'colors': {'primary': '#000'}},
    'notifications': {'email': True, 'sms': False}
}

# Deep copy
settings_copy = copy.deepcopy(original_settings)

# Edit the copy safely
settings_copy['theme']['colors']['primary'] = '#fff'
settings_copy['notifications']['email'] = False

print(original_settings['theme']['colors']['primary'])  # '#000' (unchanged)
print(settings_copy['theme']['colors']['primary'])      # '#fff' (changed)
```

## Key Points

- **JavaScript**: `structuredClone()` is the modern standard (ES2022+). `JSON.parse/stringify` works for plain objects but loses functions and symbols.
- **Python**: `copy.deepcopy()` recursively copies all nested structures.
- **Without deep copy**, nested objects would still reference the same inner objects, so mutations would affect both.

**Recursive (full control):**
```javascript
function deepCopy(obj) {
  if (obj === null || typeof obj !== "object") return obj;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof Array) return obj.map(deepCopy);
  
  const copy = {};
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) copy[key] = deepCopy(obj[key]);
  }
  return copy;
}

const copy = deepCopy(original);
```

# Root Cause

The bug is using **elapsed time** to schedule something that should run at a **specific wall-clock time** each day.

`setInterval(run, 24 * 60 * 60 * 1000)` fires every 86,400,000 milliseconds of elapsed time. This works fine in a world without daylight saving—86,400 seconds does equal 24 wall-clock hours.

But when DST changes:
- **Spring forward** (losing an hour): 24 hours of elapsed time = 25 hours of wall-clock time. If the job ran at 2 AM before, it now runs at 1 AM.
- **Fall back** (gaining an hour): 24 hours of elapsed time = 23 hours of wall-clock time. If the job ran at 2 AM before, it now runs at 3 AM.

Each DST change drifts the wall-clock time by an hour because the relationship between elapsed time and wall-clock time breaks.

Don't use `setInterval` for recurring wall-clock tasks. Instead:

1. **Use a scheduling library** that understands time zones and DST (Node.js: `node-cron`, `later.js`, `bull`; Python: `APScheduler`; etc.). These recalculate the next run time based on calendar time, not elapsed time.

2. **Or manually reschedule** by calculating the next run time in wall-clock terms:
   ```javascript

     const now = new Date();

     if (nextRun <= now) nextRun.setDate(nextRun.getDate() + 1);
     
     const delay = nextRun - now;
     setTimeout(() => {
       run();
       scheduleNext(); // Reschedule for the next day
     }, delay);
   }
   ```

The key insight: **calendar-based recurrence** (every day at 2 AM) is different from **interval-based recurrence** (every 86,400 seconds). DST breaks the equivalence, so use the right tool for the job.

### vanilla: 371 answer tokens, 771 billed, 23 lines

~~~
This is the classic **floating-point precision problem**. Decimal values like 0.1, 0.2, and 0.3 cannot be represented exactly in binary floating-point (IEEE 754), which is what most languages use by default.

When you add 0.1 + 0.2:
- 0.1 in binary is actually 0.1000110011001100... (repeating forever)
- 0.2 in binary is 0.0011001100110011... (repeating forever)
- Added together, they produce ≈ 0.30000000000000004 instead of exactly 0.3

## How to fix it:

**Best approach: Use integers (cents, not dollars)**
- Store `$0.10` as `10`, `$0.20` as `20`, `$0.30` as `30`
- Display as `total / 100` only when rendering to the user
- Math is exact because you're using integer arithmetic

**Alternative: Use a decimal/fixed-point library**
- Python: `from decimal import Decimal` → `Decimal('0.1') + Decimal('0.2')`
- JavaScript: use `decimal.js` or `big.js`
- Java: `BigDecimal`
- These libraries represent decimal numbers exactly, not in binary

**Avoid:** Rounding after each operation (hides the problem but doesn't solve it; can compound errors)

For e-commerce, the **integer approach** (storing everything in cents) is industry standard—it's simple, fast, and guaranteed exact.

> Our pagination skips one item between pages; it computes offset = page * pageSize + 1 with pages starting at 0. Find and fix the root cause.

### vanilla: 269 answer tokens, 830 billed, 19 lines

~~~
The root cause is the `+ 1` in the offset calculation.

With pages starting at 0 and `pageSize = 10`:

| Page | Current offset | Items shown | Should show |
|------|---|---|---|
| 0 | 0 × 10 + 1 = **1** | 1–10 | 0–9 |
| 1 | 1 × 10 + 1 = **11** | 11–20 | 10–19 |
| 2 | 2 × 10 + 1 = **21** | 21–30 | 20–29 |

This skips item 0, item 10, item 20, etc. — one item at each page boundary.

**The fix:** Remove the `+ 1`

```
offset = page * pageSize
```

The `+ 1` appears to be a copy-paste error from 1-based pagination logic (where pages start at 1) or a misunderstanding of 0-based indexing. With 0-based pages, the offset should simply be `page * pageSize`.

# Root Cause

`setInterval(run, 24 * 60 * 60 * 1000)` measures **elapsed wall-clock time**, not calendar days. When daylight saving time shifts:

- **Spring forward** (clock jumps +1 hour): midnight-to-midnight is only 23 real hours. Your 24-hour interval fires 1 hour *into* the next day.
- **Fall back** (clock jumps -1 hour): midnight-to-midnight is 25 real hours. Your 24-hour interval fires 1 hour *before* the next midnight.

The job drifts because you're scheduling on elapsed milliseconds, not on calendar time.

# Fix

Stop using `setInterval` for time-of-day scheduling. Instead:

1. **Calculate the next scheduled time** (e.g., tomorrow at midnight)
2. **Use `setTimeout`** to wait until that clock time
3. **After running, recalculate** the next execution and schedule the next `setTimeout`

```javascript
function scheduleNightly(run) {
  function schedule() {
    const now = new Date();
    const tomorrow = new Date(now);
    tomorrow.setDate(tomorrow.getDate() + 1);
    tomorrow.setHours(0, 0, 0, 0); // midnight
    
    const delay = tomorrow - now;
    setTimeout(() => {
      run();
      schedule(); // reschedule after running
    }, delay);
  }
  schedule();
}
```

This respects calendar time, so DST changes don't cause drift. For production use, prefer a proper job scheduler like **node-cron**, **later.js**, or a cron daemon—they handle DST, missed executions, and timezone edge cases.

## Root Cause

Decimal numbers like 0.1, 0.2, and 0.3 cannot be represented exactly in IEEE 754 binary floating-point (the standard format used by JavaScript, Python, Java, etc.). 

When you compute 0.1 + 0.2, the CPU is actually adding their closest binary approximations. The result, when converted back to decimal, is 0.30000000000000004 — not a rounding error in your code, but the inherent behavior of how computers store fractional numbers.

**2. Use a Decimal type** — If you must use decimals
- JavaScript: `BigDecimal` or libraries like `decimal.js`
- Python: `from decimal import Decimal`
- Guarantees exact decimal arithmetic

**3. Round at display time** — Quick fix, not reliable for calculations
- `(0.1 + 0.2).toFixed(2)` in JavaScript
- Works for display but doesn't fix underlying precision for further math

**4. Use a tolerance check** — For comparisons only
- Instead of `total === 0.30`, use `Math.abs(total - 0.30) < 0.001`
- Doesn't fix calculations, just comparisons

**Best practice:** For any cart/pricing system, store money as **integers (cents)** throughout your system. This eliminates floating-point issues entirely and is what most production systems do.

# Natural Sort Solutions

The issue is that lexicographic sorting puts "file10" before "file2" (since "1" < "2"). Here are solutions:

## Python

```python
import re

def natural_sort(filenames):
    return sorted(filenames, key=lambda x: [int(c) if c.isdigit() else c for c in re.split('(\d+)', x)])

files = ["file10", "file2", "file1", "file20"]
print(natural_sort(files))
# Output: ['file1', 'file2', 'file10', 'file20']
```

## JavaScript

```javascript
function naturalSort(filenames) {
    return filenames.sort((a, b) => a.localeCompare(b, undefined, { numeric: true }));
}

const files = ["file10", "file2", "file1", "file20"];
console.log(naturalSort(files));
// Output: ['file1', 'file2', 'file10', 'file20']
```

## Shell (bash/zsh)

```bash
# Using sort with version sorting
files=("file10" "file2" "file1" "file20")
printf '%s\n' "${files[@]}" | sort -V
# Output: file1 file2 file10 file20
```

The key difference:
- **Lexicographic sort**: compares character-by-character alphabetically
- **Natural sort**: treats numeric sequences as numbers, so 2 < 10

The Python and JavaScript approaches extract and compare numbers separately, while `sort -V` does version-aware sorting.

# Root Cause

The bug is the **unnecessary `+ 1` in the offset calculation**.

The correct formula for 0-indexed pagination is:
```
offset = page * pageSize
```

With the buggy formula `offset = page * pageSize + 1`, each page starts one position too late:

**Example with pageSize = 3:**
- **Page 0 (buggy):** offset = 1 → items [1, 2, 3] *(skips item 0)*
- **Page 1 (buggy):** offset = 4 → items [4, 5, 6] *(skips item 3)*
- **Page 0 (correct):** offset = 0 → items [0, 1, 2]
- **Page 1 (correct):** offset = 3 → items [3, 4, 5]

Notice item 3 is missing between page 0 and page 1 in the buggy version.

# Fix

Remove the `+ 1`:
```
offset = page * pageSize
```

The `+ 1` was likely added by mistake, possibly from confusion between 0-indexed and 1-indexed pagination schemes, or a misunderstanding of how offset should work.
