```javascript
// Parse --port and --verbose
const args = process.argv.slice(2);
const opts = {
  port: 3000,
  verbose: false
};

for (let i = 0; i < args.length; i++) {
  if (args[i] === '--port' && args[i + 1]) {
    opts.port = parseInt(args[i + 1], 10);
    i++;
  } else if (args[i] === '--verbose') {
    opts.verbose = true;
  }
}

console.log(opts);
```
