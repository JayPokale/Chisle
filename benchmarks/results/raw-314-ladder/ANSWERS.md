```javascript
const args = process.argv.slice(2);
```

Here are a few ways to convert an ISO timestamp to a localized date format:

## Native JavaScript (Recommended)

```javascript
function formatApiDate(isoString) {
  return new Date(isoString).toLocaleDateString('en-US', {
    year: 'numeric',
    month: 'short',
    day: 'numeric'
  });
}

// Usage

console.log(formatApiDate(timestamp)); // "Nov 23, 2026"
```

## In a React component

```jsx
function DateDisplay({ timestamp }) {
  const formatted = new Date(timestamp).toLocaleDateString('en-US', {
    year: 'numeric',
    month: 'short',
    day: 'numeric'
  });
  
  return <span>{formatted}</span>;
}
```

## With a library (date-fns)

```javascript
import { format } from 'date-fns';

const timestamp = "2026-11-23T14:30:00Z";
console.log(format(new Date(timestamp), 'MMM d, yyyy')); // "Nov 23, 2026"
```

## With day.js (lighter alternative)

```javascript
import dayjs from 'dayjs';

const timestamp = "2026-11-23T14:30:00Z";
console.log(dayjs(timestamp).format('MMM DD, YYYY')); // "Nov 23, 2026"
```

The native `toLocaleDateString()` approach is best if you don't need a lot of date manipulation. For more complex formatting needs, `date-fns` or `day.js` are lighter alternatives to Moment.js.

```python
def remove_duplicate_emails(emails):
    return list(dict.fromkeys(emails))
```

**JavaScript:**
```javascript
function removeDuplicateEmails(users) {
    const seen = new Set();
    const result = [];
    for (const user of users) {
        if (!seen.has(user.email)) {
            seen.add(user.email);
            result.push(user);
        }
    }
    return result;
}
```

The key approach in all versions:
1. Use a **set** to track emails we've already seen (O(1) lookup)
2. Iterate through the list once
3. Keep only items whose emails haven't been seen before
4. Add the email to the set as we go

This maintains order and keeps the first occurrence of each email with **O(n)** time complexity.

```javascript
function deepCopy(obj) {
  if (obj === null || typeof obj !== 'object') return obj;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof Array) return obj.map(deepCopy);
  if (obj instanceof Object) {
    const copy = {};
    for (let key in obj) {
      if (obj.hasOwnProperty(key)) {
        copy[key] = deepCopy(obj[key]);
      }
    }
    return copy;
  }
}
```
